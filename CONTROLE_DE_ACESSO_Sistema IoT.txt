// Arduino UNO, HC-SR04, microservo, buzzer, LCD 16x2 I2C,
// LED verde, LED vermelho (resistores de 220 ohm) e push-button.

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Servo.h>

// ---------- Pinos ----------
const byte PINO_BOTAO = 2;
const byte PINO_LED_VERDE = 4;
const byte PINO_LED_VERMELHO = 5;
const byte PINO_SERVO = 6;
const byte PINO_BUZZER = 8;
const byte PINO_TRIG = 9;
const byte PINO_ECHO = 10;
// LCD I2C: SDA = A4, SCL = A5

// ---------- Parametros configuraveis ----------
const int DIST_LIMITE_CM = 20;
const byte LEITURAS_CONFIRMACAO = 3;
const unsigned long INTERVALO_LEITURA_MS = 60;
const unsigned long TEMPO_ALERTA_MS = 5000;
const unsigned long TEMPO_ALERTA_AUTORIZADO_MS = 10000;
const unsigned long TEMPO_NORMALIZACAO_MS = 1500;
const unsigned long TEMPO_MIN_ABERTO_MS = 3000;
const int ANGULO_FECHADO = 0;
const int ANGULO_ABERTO = 90;

const unsigned long DEBOUNCE_MS = 50;
const unsigned long TIMEOUT_ECO_US = 25000; // espera maxima pelo eco
const unsigned int FREQ_ALARME_HZ = 2000;

// Endereco 0x20 (32) no Tinkercad. LCD fisico: geralmente 0x27 ou 0x3F.
LiquidCrystal_I2C lcd(0x20, 16, 2);
Servo cancela;

// ---------- Estados ----------
enum Estado { FECHADO, ACESSO, ALERTA };
Estado estado = FECHADO;

unsigned long ultimaLeitura = 0;
unsigned long inicioAbertura = 0;
unsigned long inicioPresenca = 0;
unsigned long ultimaPresenca = 0;
byte leiturasSeguidas = 0;
bool presencaConfirmada = false;
bool presencaEmCurso = false;
bool autorizado = false; // liberacao pelo botao neste ciclo
unsigned int totalEventos = 0;

bool leituraAnterior = HIGH;
bool botaoEstavel = HIGH;
unsigned long mudancaBotao = 0;

char linhaAtual[2][17] = {"", ""};

// ---------- LCD (so reescreve quando o texto muda) ----------
void escreverLinha(byte linha, const char* texto) {
char buf[17];
snprintf(buf, sizeof(buf), "%-16s", texto);
if (strcmp(buf, linhaAtual[linha]) != 0) {
lcd.setCursor(0, linha);
lcd.print(buf);
strcpy(linhaAtual[linha], buf);
}
}

unsigned long limiteAlerta() {
return autorizado ? TEMPO_ALERTA_AUTORIZADO_MS : TEMPO_ALERTA_MS;
}

void atualizarLCD(unsigned long agora) {
char texto[17];

if (estado == FECHADO) {
escreverLinha(0, "Acesso Fechado");
snprintf(texto, sizeof(texto), "Eventos: %u", totalEventos);
escreverLinha(1, texto);
} else if (estado == ACESSO) {
escreverLinha(0, autorizado ? "Cracha Liberado" : "Acesso Detectado");
if (presencaEmCurso) {
unsigned long decorrido = agora - inicioPresenca;
unsigned long restante = 0;
if (decorrido < limiteAlerta()) {
restante = (limiteAlerta() - decorrido + 999) / 1000;
}
snprintf(texto, sizeof(texto), "Alerta em: %lus", restante);
} else {
snprintf(texto, sizeof(texto), "Evento: %u", totalEventos);
}
escreverLinha(1, texto);
} else {
escreverLinha(0, "ALERTA:");
escreverLinha(1, "PORTA ABERTA!");
}
}

// ---------- Registro na porta serial ----------
void registrarEvento(const char* descricao) {
Serial.print("[");
Serial.print(millis());
Serial.print(" ms] Evento ");
Serial.print(totalEventos);
Serial.print(": ");
Serial.println(descricao);
}

// ---------- Mudanca de estado ----------
void mudarEstado(byte novo, unsigned long agora) {
estado = (Estado)novo;

if (estado == FECHADO) {
noTone(PINO_BUZZER);
cancela.write(ANGULO_FECHADO);
digitalWrite(PINO_LED_VERDE, LOW);
digitalWrite(PINO_LED_VERMELHO, HIGH);
autorizado = false;
presencaEmCurso = false;
registrarEvento("FECHADO");
} else if (estado == ACESSO) {
totalEventos++;
inicioAbertura = agora;
ultimaPresenca = agora;
cancela.write(ANGULO_ABERTO);
digitalWrite(PINO_LED_VERDE, HIGH);
digitalWrite(PINO_LED_VERMELHO, LOW);
registrarEvento(autorizado ? "ACESSO (cracha)" : "ACESSO (sensor)");
} else {
tone(PINO_BUZZER, FREQ_ALARME_HZ); // alarme continuo
digitalWrite(PINO_LED_VERDE, LOW);
digitalWrite(PINO_LED_VERMELHO, HIGH);
registrarEvento("ALERTA: PORTA ABERTA!");
}

atualizarLCD(agora);
}

// ---------- Sensor ----------
long lerDistancia() {
digitalWrite(PINO_TRIG, LOW);
delayMicroseconds(2);
digitalWrite(PINO_TRIG, HIGH);
delayMicroseconds(10);
digitalWrite(PINO_TRIG, LOW);
unsigned long duracao = pulseIn(PINO_ECHO, HIGH, TIMEOUT_ECO_US);
if (duracao == 0) return -1; // sem eco
return duracao * 0.034 / 2;
}

// Confirma presenca apos leituras consecutivas abaixo do limite
void lerSensor() {
long distancia = lerDistancia();
if (distancia >= 0 && distancia < DIST_LIMITE_CM) {
if (leiturasSeguidas < LEITURAS_CONFIRMACAO) leiturasSeguidas++;
} else {
leiturasSeguidas = 0;
}
presencaConfirmada = (leiturasSeguidas >= LEITURAS_CONFIRMACAO);
}

// ---------- Botao (cracha) ----------
bool botaoPressionado(unsigned long agora) {
bool leitura = digitalRead(PINO_BOTAO); // apertado = LOW
if (leitura != leituraAnterior) {
mudancaBotao = agora;
leituraAnterior = leitura;
}
if (agora - mudancaBotao >= DEBOUNCE_MS && leitura != botaoEstavel) {
botaoEstavel = leitura;
return botaoEstavel == LOW;
}
return false;
}

void tratarBotao(unsigned long agora) {
if (estado == FECHADO) {
autorizado = true;
mudarEstado(ACESSO, agora);
} else if (estado == ACESSO && !autorizado) {
autorizado = true;
registrarEvento("Cracha: tempo estendido");
}
// No estado ALERTA o botao e ignorado
}

// ---------- Logica ----------
void atualizarEstado(unsigned long agora) {
if (estado == FECHADO) {
if (!presencaConfirmada) return;
mudarEstado(ACESSO, agora);
}

if (presencaConfirmada) {
ultimaPresenca = agora;
if (!presencaEmCurso) {
presencaEmCurso = true;
inicioPresenca = agora;
}
}

bool normalizado = !presencaConfirmada &&
agora - ultimaPresenca >= TEMPO_NORMALIZACAO_MS;

if (estado == ACESSO) {
if (normalizado) {
presencaEmCurso = false;
if (agora - inicioAbertura >= TEMPO_MIN_ABERTO_MS) {
mudarEstado(FECHADO, agora);
}
} else if (presencaConfirmada && agora - inicioPresenca >= limiteAlerta()) {
mudarEstado(ALERTA, agora);
}
} else if (estado == ALERTA && normalizado) {
mudarEstado(FECHADO, agora);
}
}

// ---------- Setup / Loop ----------
void setup() {
Serial.begin(9600);
pinMode(PINO_TRIG, OUTPUT);
pinMode(PINO_ECHO, INPUT);
pinMode(PINO_BUZZER, OUTPUT);
pinMode(PINO_LED_VERDE, OUTPUT);
pinMode(PINO_LED_VERMELHO, OUTPUT);
pinMode(PINO_BOTAO, INPUT_PULLUP);

cancela.attach(PINO_SERVO);
lcd.init();
lcd.backlight();

mudarEstado(FECHADO, millis());
}

void loop() {
unsigned long agora = millis();

if (botaoPressionado(agora)) tratarBotao(agora);

if (agora - ultimaLeitura >= INTERVALO_LEITURA_MS) {
ultimaLeitura = agora;
lerSensor();
atualizarEstado(agora);
atualizarLCD(agora);
}
}
